## 6.1.1 規則不要用 getters or setters

- 非布林值不要用 getters or setters
- getter破壞封裝好的物件，︁變成全域任何人都可以呼叫，可能會有預期之外的用法
- setter 破壞封裝，增加耦合性

- 拉式架構
``` javascript
class Website {
    Constructor(private url: string){}
    getUrl(){
        return this.url;
    }
}
class User {
    constructor(private username: string){}
    getWebsite(){
        return this.username;
    }
}
class BlogPost {
    constructor(private author: User, private id: string){}
    getAuthor(){
        return this.author;
    }
    getId(){
        return this.id;
    }
}
function generatePostLink(website: Website, post: BlogPost){
    let url = website.getUrl();
    let user = post.getAuthor();
    let name = user.getWebsite();
    let postId = post.getId();
    return url + name + postId;
    
}
```
- 推式架構
```javascript=
class Website {
  constructor(private url: string) {}
  generatePostLink(name: string, postId: string) {
    return this.url + name + postId;
  }
}
class User {
  constructor(private username: string) {}
  generatePostLink(website: Website, postId: string) {
    return website.generatePostLink(this.username, postId);
  }
}
class BlogPost {
  constructor(private author: User, private id: string) {}
  generatePostLink(website: Website) {
    return this.author.generatePostLink(website, this.id);
  }
}
function generatePostLink(website: Website, post: BlogPost) {
  return post.generatePostLink(website);
}
```
- 拉式架構
